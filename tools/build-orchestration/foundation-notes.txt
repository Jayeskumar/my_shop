// modules/catalog/index.ts
import { z } from 'zod'
import { newId } from '@kadai/shared'
import { dataOf, listOf, PageQuerySchema, zId } from '@kadai/shared/api/common'
import { type KadaiPlugin, requirePermission, requireFeature, parsePage, paginate, idempotentTx, route } from '../../core/http'
import { assertWithinLimit } from '../../core/entitlements'
import { audit } from '../../core/audit'
import { notFound } from '../../core/errors'
import { defineModule } from '../module'

const BrandSchema = z.object({ id: zId, name: z.string(), is_active: z.boolean() })

const routes: KadaiPlugin = async (app) => {                 // registered under /api/v1, auth = tenant
  app.get('/brands', {
    schema: { querystring: PageQuerySchema.extend({ q: z.string().optional() }), response: { 200: listOf(BrandSchema) } },
    preHandler: [requirePermission('catalog.read')],          // any-of → 403 FORBIDDEN {required}
  }, async (req) => {
    const page = parsePage(req.query)
    return req.tx((tx) => {                                   // ONE RLS transaction for req.ctx
      let q = tx.selectFrom('brands').select(['id', 'name', 'is_active']).where('tenant_id', '=', req.ctx.tenantId)
      if (req.query.q) q = q.where('name', 'ilike', `%${req.query.q}%`)
      return paginate(q.orderBy('name'), page)                // → { data, page: { limit, offset, total } }
    })
  })
  app.post('/brands', {
    schema: { body: z.object({ name: z.string().trim().min(1) }), response: { 201: dataOf(BrandSchema) } },
    preHandler: [requirePermission('catalog.write'), requireFeature('catalog')],   // 402 FEATURE_NOT_IN_PLAN
  }, async (req, reply) => idempotentTx(req, reply, async (tx) => {             // Idempotency-Key header or body client_ref
    await assertWithinLimit(tx, req.ctx, 'products')          // 402 PLAN_LIMIT_REACHED (only meaningful for limit metrics)
    const row = await tx.insertInto('brands').values({ id: newId(), tenant_id: req.ctx.tenantId, name: req.body.name })
      .returning(['id', 'name', 'is_active']).executeTakeFirstOrThrow()
    await audit(tx, req.ctx, { action: 'brand.create', entity_type: 'brand', entity_id: row.id, summary: row.name })
    return { data: row }                                      // default status 201; pass { status } to change
  }))
  // A shared catalogue entry (defineRoutes in @kadai/shared/api/catalog): route(app, catalogRoutes.getBrand, async (req) => ({ data: … }))
}
export const catalogModule = defineModule({ name: 'catalog', routes /*, publicRoutes, webhooks, cliCommands */ })

Route conventions:
- Throw AppError (or a helper) for errors. pg errors are mapped automatically: 23505 → 409 DUPLICATE {constraint}, 23503/23514 → 422, 42501 → 403, P0001 'append-only' → 409.
- Route config flags:
  - `config: { public: true }` — no authentication at all;
  - `config: { auth: 'user' }` — tenant-less tokens allowed: select-tenant, logout, password change, platform admin with requirePlatformAdmin();
  - `config: { auth: 'optional' }`;
  - `config: { allowSuspended: true }` — billing / self-service;
  - `config: { allowReadOnly: true }` — license upload, logout;
  - `config: { rateLimit: RATE_LIMITS.login | otp | webhook | public }`.
- publicRoutes are registered under /public/v1 and webhooks under /webhooks. Webhooks read `req.rawBody` (a Buffer) for JSON, form-encoded and text bodies.
- Uploads: @fastify/multipart is registered globally; call `await req.file()`, then `app.kadai.files.saveFile(tx, req.ctx, { purpose, filename, content_type, data })`.
- `req.ctx` throws when the route has no tenant context; check `req.hasCtx` first. `req.auth` is either { kind: 'user', userId, tenantId|null, membershipId|null, sessionId, familyId, deviceId, amr } or { kind: 'api_key', apiKeyId, tenantId, keyPrefix }. `req.server.kadai` / `app.kadai` is the KadaiDeps object.

  defineJob({ name: 'payments.expire-intents', cron: '* * * * *', tz: 'Asia/Kolkata', handler: async (_j, deps) => { /* fan out: deps.db.systemDb tenants */ } }),

Job rules:
- Names must match `module.job-name`, be unique, and the registry validates them.
- Enqueue from anywhere: `enqueue(name, data, { tx?, startAfter?, singletonKey?, singletonSeconds?, … })` from core/jobs. Passing `tx` makes the job exist only if the business transaction commits.
- Per-job worker options: `work: { localConcurrency, pollingIntervalSeconds }`.
- core.cleanup runs hourly (cron '17 * * * *', IST).
- The worker heartbeat is written to system_settings 'worker_heartbeat' every 30 s.

===== DOMAIN SERVICES + AUTH (apps/server/src/core, modules/auth) =====
IMPORTS

GENERAL RULES
- All services take (tx, ctx, input) and run inside the caller's transaction: req.tx(...) or withTenant(appDb, ctx, ...).
- They never open their own transactions, so stock, journal and payment postings in one request are atomic.
- ctx needs tenantId and userId; payments also needs storeIds, ip, userAgent and requestId. A full RequestContext or systemContext(tenantId) works.

NUMBERING
`nextNumber(tx, ctx, { docType: NumberSeriesDocType, storeId, registerId?, date: 'YYYY-MM-DD' })` → `{ number, fy, seq, series_id }`
- Call it in the transaction that creates the document. Example: `await nextNumber(tx, req.ctx, { docType: 'sales_invoice', storeId, registerId: req.ctx.registerId, date: invoice_date })` → 'S1-A/2627/00042'.
- Pass registerId only for POS sales_invoice or bill_of_supply. Without it you get the back-office series 'S1/2627/…'.
- Errors: CONFLICT when the prefix is too long; NUMBER_SERIES_EXHAUSTED (409).

`peekSeries(tx, ctx, sameInput)` → `{ series_id|null, prefix, next_number, padding, fy, next }`
- No lock and no row creation. Use it for POS bootstrap.

`acceptOfflineNumber(tx, ctx, { docType, storeId, registerId, date, number })` → `{ number, fy, seq, series_id, renumbered, offline_number }`
- Offline sync must be in creation order. Audit when renumbered is true.

INVENTORY
`postStock(tx, ctx, movements: StockMovementInput[], { allowNegative? })` → `PostedMovement[]` in the same order as the input.
- Input: `{ store_id, product_id, batch_id?, qty_milli (signed, ≠0), movement_type, unit_cost_paise? (per pack excl. tax; required for opening/purchase/transfer_in), ref_type?, ref_id?, ref_line_id?, reason? }`.
- Each result: `{ movement_id, unit_cost_paise, cost_paise (|value|), value_paise (signed), balance_after_milli, avg_cost_after_paise, skipped }`.
- Sale COGS for a line = Σ result.cost_paise.
- Example: `postStock(tx, ctx, [{ store_id, product_id, batch_id, qty_milli: -2000, movement_type: 'sale', ref_type: 'sales_invoice', ref_id: inv.id, ref_line_id: line.id }])`.
- Signs per type: sale/transfer_out/damage/expiry_writeoff/purchase_return/purchase_cancel < 0; opening/purchase/sale_return/sale_cancel/transfer_in > 0; adjustment/stock_take ±.
- Offline sync passes `{ allowNegative: true }`.

`getAvailable(tx, ctx, { store_id, product_id, batch_id? })` → milli
- batch_id undefined = all batches; null = the row without a batch.

`pickBatchesFefo(tx, ctx, { store_id, product_id, qty_milli, today: todayIn(tz), allowNegative? })` → `[{ batch_id|null, qty_milli, batch_no, expiry_date }]`
- Feed the result straight into postStock. Non-batch products return a single pick with batch_id null.

LEDGER
`postJournal(tx, ctx, { entry_date, store_id?, source_type: JournalSourceType, source_id?, narration?, number?, lines: [{ account: {subtype} | {id}, debit_paise?, credit_paise?, party?: {type, id}, memo?, store_id? }] })` → `{ entry_id: string | null }`
- Example: `postJournal(tx, ctx, { entry_date: d, store_id, source_type: 'sales_invoice', source_id: inv.id, lines: [{ account: { subtype: 'receivable' }, debit_paise: total, party: customer ? { type: 'customer', id } : null }, { account: { subtype: 'sales' }, credit_paise: taxable }, …, { account: { subtype: 'round_off' }, credit_paise|debit_paise }] })`.
- Unbalanced entries → 422 UNPROCESSABLE {debit_paise, credit_paise, difference_paise}.
- Never update balance_paise yourself; postJournal is the only writer.

`reverseJournal(tx, ctx, { source_type, source_id, entry_date, narration? })` → `{ entry_id|null, entry_ids }`
- A second call → CONFLICT.

`systemAccountId(tx, ctx, 'cash' | … )` → id (cached per transaction).

PAYMENTS
`recordPayment(tx, ctx, input)` → PaymentRecord (payments row plus allocations[] plus replayed).
- Input: `{ direction: 'in'|'out', store_id, register_id?, shift_id? (must be open; sets register), method, amount_paise, payment_date, party?: {type, id}, reference?, payment_intent_id?, account_id?, allocations?: [{ target_type, target_id, amount_paise }], notes?, client_ref?, number?: boolean, paid_at? }`.
- POS tender example: `recordPayment(tx, ctx, { direction: 'in', store_id, register_id, shift_id, method: 'upi', amount_paise, payment_date, party, payment_intent_id, allocations: [{ target_type: 'sales_invoice', target_id: inv.id, amount_paise }] })`.
- The invoice row must already be status 'completed' with due_paise = total_paise. Allocations update paid_paise, due_paise and payment_status.
- Supplier payment: `direction: 'out', party: {type: 'supplier', id}, allocations: purchase_bill`, with the bill status 'posted'.
- Credit-note settlement:
  - cash/upi: `direction: 'out'` allocated to the credit_note (updates settled_paise);
  - store credit: `method: 'store_credit', direction: 'out'`;
  - redeeming store credit: `method: 'store_credit', direction: 'in'` (checked against storeCreditBalance).
- payment_intent_id requires status 'paid', unused, the same store and an amount equal to paid_amount_paise. The intent and the payment are linked both ways.

`cancelPayment(tx, ctx, { payment_id, reason, date? })` → `{ reversal_entry_id }`
- Sales/purchase cancellations should cancel their payments.

AUTH AND SESSION BEHAVIOUR THE WEB DEPENDS ON
- Session endpoints return `{data: SessionResponse}`: signup 201, login 200, otp/verify (login purpose), select-tenant, refresh, pin-login, setup 201. They set the cookie kadai_rt (HttpOnly; SameSite=Strict; Path=/api/v1/auth; Secure only when PUBLIC_BASE_URL is https).
- The body contains refresh_token only with the `X-Kadai-Token-Delivery: body` header, or when refresh was called with a body refresh_token.
- tenant_id is set when the user has exactly one active membership, otherwise null. /me on a tenant-less token → 403 FORBIDDEN with details.reason 'tenant_selection_required'; the client then calls select-tenant, which rotates the session (the old cookie becomes a reused token).
- Cookie refresh needs `X-Requested-With: kadai` (otherwise 403 reason 'csrf').
  - Reuse, logout or lost access → 401 SESSION_REVOKED and the cookie is cleared.
  - A revoked device → 403 DEVICE_NOT_TRUSTED.
- Logout `{all?}` → 204 and clears the cookie; it also works with only the cookie.
- password/change → 204; a wrong current password is 400 VALIDATION_ERROR on current_password.
- password/reset → 204 (single-use token, all sessions revoked).
- Lockouts: 423 ACCOUNT_LOCKED / PIN_LOCKED with details.retry_after_seconds and a Retry-After header.
- OTP verify: wrong code → 400 OTP_INVALID {attempts_left}; expired or used → OTP_EXPIRED. Signup and reset_password return `{data: {otp_token, expires_in}}`.
- GET /me → MeResponse: permissions expanded; store-scoped stores and registers including inactive ones; entitlements DTO; settings with defaults; mode; device for PIN sessions; server_time. A suspended tenant gets 403 TENANT_SUSPENDED.
- /setup/status and /setup exist only on-prem (404 in cloud). token_required is true when KADAI_SETUP_TOKEN is set.
- Identity must create device secrets with generateDeviceSecret() from core/tokens and store `hash` in devices.device_secret_hash.
- After membership, role or device changes, call notifyAccessChanged. Login turns 'invited' memberships into 'active'.
